package plugin

import (
    "bytes"
    "crypto/hmac"
    "crypto/sha256"
    "encoding/base64"
    "encoding/json"
    "net/http"
    "os"
    "testing"
)

func TestActualDotNetPluginContract(t *testing.T) {
    raw, err := os.ReadFile(os.Getenv("JELLYTRACK_CONTRACT_OUTPUT"))
    if err != nil { t.Fatal(err) }
    decoder := json.NewDecoder(bytes.NewReader(raw))
    decoder.UseNumber()
    var payloads []map[string]any
    if err := decoder.Decode(&payloads); err != nil { t.Fatal(err) }
    db, handler := testServer(t)
    var stored string
    if err := db.QueryRow(`SELECT "pluginApiKey" FROM "GlobalSettings" WHERE "id"='global'`).Scan(&stored); err != nil { t.Fatal(err) }
    serverID := "contract-jellyfin"
    mac := hmac.New(sha256.New, []byte(stored))
    mac.Write([]byte(keyPepperContext)); mac.Write([]byte{0}); mac.Write([]byte(serverID))
    scopedKey := "jts4." + base64.RawURLEncoding.EncodeToString([]byte(serverID)) + "." + base64.RawURLEncoding.EncodeToString(mac.Sum(nil))
    for _, payload := range payloads {
        response := send(t, handler, scopedKey, payload)
        if response.Code != http.StatusOK { t.Fatalf("%s: %d %s", payload["event"], response.Code, response.Body) }
        t.Logf("accepted %s %v", payload["event"], payload["changeType"])
    }
    var histories, pauses, seeks, audio, subtitles int
    if err := db.QueryRow(`SELECT COUNT(*),SUM("pauseCount"),SUM("seekCount"),SUM("audioChanges"),SUM("subtitleChanges") FROM "PlaybackHistory"`).Scan(&histories, &pauses, &seeks, &audio, &subtitles); err != nil { t.Fatal(err) }
    if histories != 1 || pauses != 1 || seeks != 1 || audio != 1 || subtitles != 1 {
        t.Fatalf("history/resume/telemetry mismatch: histories=%d pause=%d seek=%d audio=%d subtitles=%d", histories, pauses, seeks, audio, subtitles)
    }
    var active, users, media int
    if err := db.QueryRow(`SELECT COUNT(*) FROM "ActiveStream"`).Scan(&active); err != nil { t.Fatal(err) }
    if err := db.QueryRow(`SELECT COUNT(*) FROM "User" WHERE "jellyfinUserId"='11111111-1111-1111-1111-111111111111'`).Scan(&users); err != nil { t.Fatal(err) }
    if err := db.QueryRow(`SELECT COUNT(*) FROM "Media" WHERE "jellyfinMediaId"='22222222-2222-2222-2222-222222222222'`).Scan(&media); err != nil { t.Fatal(err) }
    if active != 0 || users != 1 || media != 1 { t.Fatalf("identity/session cleanup mismatch: active=%d users=%d media=%d", active, users, media) }
}
